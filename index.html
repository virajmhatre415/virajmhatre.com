function initStatsCounter() { const statNumbers = document.querySelectorAll('.stat-number'); let animated = false; function checkScroll() { if (animated) return; const heroStats = document.querySelector('.hero-stats-row'); if (!heroStats) return; const rect = heroStats.getBoundingClientRect(); if (rect.top <= window.innerHeight) { animated = true; statNumbers.forEach(el => { const raw = el.getAttribute('data-value'); const num = parseFloat(raw); const suffix = el.getAttribute('data-suffix') || ''; if (isNaN(num)) return; let cur = 0; const duration = 1200; const steps = 30; const inc = num / steps; const timer = setInterval(() => { cur += inc; if (cur >= num) { cur = num; clearInterval(timer); } el.textContent = Math.floor(cur) + suffix; }, duration / steps); }); } } window.addEventListener('scroll', checkScroll); checkScroll(); }
